/** * lamastudio.pk - Clean Multi-Tenant Cloud ERP Backend API Bridge * Target Master Sheet: "SchoolsMaster" */ const SHEET_NAME = "SchoolsMaster"; function doPost(e) { try { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) { sheet = ss.insertSheet(SHEET_NAME); sheet.appendRow([ "Timestamp", "schoolName", "schoolCodeId", "adminEmail", "adminPassword", "schoolSector", "schoolDistrict", "schoolAddress", "schoolLevel", "schoolGender", "totalStrength", "totalStaff", "principalName", "principalPhone", "institutionCode", "receiptNote", "packageTier", "status" ]); } const data = JSON.parse(e.postData.contents); sheet.appendRow([ new Date().toISOString(), data.schoolName || "", data.schoolCodeId || "", data.adminEmail || "", data.adminPassword || "", data.schoolSector || "Private", data.schoolDistrict || "", data.schoolAddress || "", data.schoolLevel || "", data.schoolGender || "", data.totalStrength || "", data.totalStaff || "", data.principalName || "", data.principalPhone || "", data.institutionCode || "", data.receiptNote || "", data.packageTier || "", "Active" ]); return ContentService .createTextOutput(JSON.stringify({ status: "success" })) .setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService .createTextOutput(JSON.stringify({ status: "error", message: error.toString() })) .setMimeType(ContentService.MimeType.JSON); } } function doGet(e) { try { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) { return ContentService.createTextOutput(JSON.stringify([])).setMimeType(ContentService.MimeType.JSON); } const rows = sheet.getDataRange().getValues(); if (rows.length <= 1) { return ContentService.createTextOutput(JSON.stringify([])).setMimeType(ContentService.MimeType.JSON); } const headers = rows[0]; const schoolList = []; for (let i = 1; i < rows.length; i++) { const row = rows[i]; schoolList.push({ "Timestamp": row[0], "schoolName": row[1], "schoolCodeId": row[2], "adminEmail": row[3], "adminPassword": row[4], "schoolSector": row[5], "schoolDistrict": row[6], "schoolAddress": row[7], "schoolLevel": row[8], "schoolGender": row[9], "totalStrength": row[10], "totalStaff": row[11], "principalName": row[12], "principalPhone": row[13], "institutionCode": row[14], "receiptNote": row[15], "packageTier": row[16], "status": row[17] }); } // Check if a specific ID lookup was requested (e.g., ?action=get&id=...) const action = e && e.parameter && e.parameter.action; const targetId = e && e.parameter && e.parameter.id; if (action === "get" && targetId) { const found = schoolList.find(s => s.schoolCodeId === targetId || s.adminEmail === targetId); return ContentService .createTextOutput(JSON.stringify(found || null)) .setMimeType(ContentService.MimeType.JSON); } // Default action: Return all schools return ContentService .createTextOutput(JSON.stringify(schoolList)) .setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService .createTextOutput(JSON.stringify({ status: "error", message: error.toString() })) .setMimeType(ContentService.MimeType.JSON); } }
